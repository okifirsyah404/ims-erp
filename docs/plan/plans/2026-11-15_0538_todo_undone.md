## 0. How the numbers were measured

```bash
# schema-qualified: only counts a table as built when it lives in the module's own schema file
bun -e '
const fs=require("fs");
const dbml=fs.readFileSync("docs/database/docs.dbml","utf8");
const blocks=[...dbml.matchAll(/^Table ([a-z_]+)\.([a-z_0-9]+)/gm)].map(m=>({schema:m[1],table:m[2]}));
const shipped={};
for(const f of fs.readdirSync("apps/backend/drizzle/schemas")){
  const s=fs.readFileSync("apps/backend/drizzle/schemas/"+f,"utf8");
  for(const m of s.matchAll(/([A-Za-z0-9_$]+)Schema\.table\(\s*["\x27`]([a-z_0-9]+)["\x27`]/g)) shipped[m[2]]=f;
}
const per={};
for(const b of blocks){ per[b.schema]=per[b.schema]||{total:0,missing:[]}; per[b.schema].total++; if(!shipped[b.table]) per[b.schema].missing.push(b.table); }
for(const [k,v] of Object.entries(per)) console.log(k,"total="+v.total,"missing="+v.missing.length);
'
# -> 274 table blocks, 145 missing
bun docs/database/validate-dbml.ts
# -> parsed: 274 tables, 235 refs / cross-module refs: 0 / timestamp gaps: 0 / drizzle parity drift: 0 / OK
```
